/* 库存统计快照表：普通库存统计生成结果持久化（第一期） */
IF NOT EXISTS (
  SELECT 1 FROM sys.objects
  WHERE object_id = OBJECT_ID(N'[dbo].[UB_ERP_Stock_stats_snapshot]') AND type IN (N'U')
)
BEGIN
  CREATE TABLE [dbo].[UB_ERP_Stock_stats_snapshot] (
    [id] INT IDENTITY(1, 1) NOT NULL,
    [start_date] DATE NOT NULL,
    [end_date] DATE NOT NULL,
    [warehouse_code] NVARCHAR(200) NOT NULL,
    [warehouse_name] NVARCHAR(500) NULL,
    [material_filter] NVARCHAR(300) NULL,
    [report_kind] NVARCHAR(50) NOT NULL CONSTRAINT [DF_Stock_stats_snapshot_kind] DEFAULT (N'normal'),
    [generated_at] DATETIME NOT NULL CONSTRAINT [DF_Stock_stats_snapshot_at] DEFAULT (GETDATE()),
    [generated_uid] NVARCHAR(50) NULL,
    [generated_uname] NVARCHAR(100) NULL,
    [generated_truename] NVARCHAR(100) NULL,
    [row_count] INT NOT NULL CONSTRAINT [DF_Stock_stats_snapshot_rows] DEFAULT (0),
    [filter_json] NVARCHAR(MAX) NULL,
    [pass] NVARCHAR(10) NOT NULL CONSTRAINT [DF_Stock_stats_snapshot_pass] DEFAULT (N'1'),
    [del] NVARCHAR(10) NOT NULL CONSTRAINT [DF_Stock_stats_snapshot_del] DEFAULT (N'0'),
    CONSTRAINT [PK_UB_ERP_Stock_stats_snapshot] PRIMARY KEY CLUSTERED ([id] ASC)
  );
  CREATE NONCLUSTERED INDEX [IX_Stock_stats_snapshot_gen] ON [dbo].[UB_ERP_Stock_stats_snapshot] ([generated_at] DESC, [id] DESC);
END
GO
